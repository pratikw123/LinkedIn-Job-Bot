def clean_text(text):
text = re.sub(r”[^a-zA-Z0-9\s.,]”, “ “, text)
return re.sub(r”\s+”, “ “, text).strip().lower()

def normalize_keyword(kw, threshold=0.62):
kw_vec = model.encode(kw, convert_to_tensor=True)
sims   = util.cos_sim(kw_vec, canonical_embeddings)[0]
idx    = int(sims.argmax())
if sims[idx].item() >= threshold:
return CANONICAL_SKILLS[idx]
return None

def extract_structured_keywords(text):
doc = nlp(text)
candidates = set()
for chunk in doc.noun_chunks:
phrase = chunk.text.lower().strip()
if 1 <= len(phrase) <= 40:
candidates.add(phrase)
for token in doc:
if token.pos_ in {“PROPN”, “NOUN”} and not token.is_stop:
candidates.add(token.text.lower())
normalized = set()
for kw in candidates:
norm = normalize_keyword(kw)
if norm:
normalized.add(norm)
return normalized

def fuzzy_match_keywords(set1, set2, threshold=0.7):
matched = set()
for kw1 in set1:
for kw2 in set2:
sim = util.cos_sim(model.encode(kw1), model.encode(kw2)).item()
if sim > threshold:
matched.add(kw1)
break
return matched

def chunk_text(text, max_words=150):
words = text.split()
return [” “.join(words[i:i+max_words]) for i in range(0, len(words), max_words)]

            if href and "/jobs/view/" in href:
                job_urls.add(href.split("?")[0])
        cur_h = driver.execute_script("return arguments[0].scrollTop", scrollable)
        if cur_h == last_h:
            break
        last_h = cur_h

    # Next page

        nxt = driver.find_element(By.CSS_SELECTOR, "button.jobs-search-pagination__button--next")
        if not nxt.is_enabled():
            break
        driver.execute_script("arguments[0].click();", nxt)
        time.sleep(3)
        page_num += 1
    except Exception:
        break

print(f"  🔍 {len(job_urls)} URLs collected for '{job_title}'")
return job_urls
